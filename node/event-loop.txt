const fs = require('fs');

console.log('1: Start');

setTimeout(() => {
    console.log('2: Timeout');
}, 0);

setImmediate(() => {
    console.log('3: Immediate');
});

fs.readFile(__filename, () => {
    setTimeout(() => console.log('4: File Timeout'), 0);
    setImmediate(() => console.log('5: File Immediate'));
    process.nextTick(() => console.log('6: File nextTick'));
});

Promise.resolve().then(() => {
    console.log('7: Promise');
});

process.nextTick(() => {
    console.log('8: nextTick');
});

console.log('9: End');

/***

1: Start
9: End
8: nextTick
7: Promise
2: Timeout
3: Immediate
6: File nextTick
5: File Immediate
4: File Timeout

***/
